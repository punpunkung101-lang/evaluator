cmake_minimum_required(VERSION 3.16)
project(evaluator_backend LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SQLite3 REQUIRED)
add_executable(evaluator_backend src/main.cpp)
target_link_libraries(evaluator_backend PRIVATE SQLite::SQLite3 ws2_32 bcrypt)

option(EVALUATOR_BUILD_TESTS "Build backend tests (db_compat_test)" OFF)
if(EVALUATOR_BUILD_TESTS)
  add_executable(db_compat_test ../tests/backend/db_compat_test.cpp)
  target_link_libraries(db_compat_test PRIVATE SQLite::SQLite3)
endif()
